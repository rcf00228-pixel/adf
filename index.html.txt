    function applyAppleIcon(emoji, bgColor) {
      const canvas = document.createElement('canvas');
      canvas.width = 512;
      canvas.height = 512;
      const ctx = canvas.getContext('2d');

      // 배경 색상 채우기
      ctx.fillStyle = bgColor || '#4f46e5';
      ctx.fillRect(0, 0, 512, 512);

      // 중앙 이모지 그리기
      ctx.font = '280px -apple-system, BlinkMacSystemFont, "Apple Color Emoji", sans-serif';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText(emoji || '📚', 256, 275);

      const dataUrl = canvas.toDataURL('image/png');

      // 기존 아이콘 태그 제거 후 새로 주입
      document.querySelectorAll("link[rel*='icon']").forEach(el => el.remove());

      const appleLink = document.createElement('link');
      appleLink.rel = 'apple-touch-icon';
      appleLink.href = dataUrl;
      document.head.appendChild(appleLink);

      const favLink = document.createElement('link');
      favLink.rel = 'icon';
      favLink.type = 'image/png';
      favLink.href = dataUrl;
      document.head.appendChild(favLink);
    }

    applyAppleIcon(savedIcon.emoji, savedIcon.bg);

    // 날짜 유틸 함수
    function toDateStr(dateObj) {
      const y = dateObj.getFullYear();
      const m = String(dateObj.getMonth() + 1).padStart(2, '0');
      const d = String(dateObj.getDate()).padStart(2, '0');
      return `${y}-${m}-${d}`;
    }

    function parseDateStr(str) {
      const [y, m, d] = str.split('-').map(Number);
      return new Date(y, m - 1, d);
    }

    function addDays(dateStr, days) {
      const d = parseDateStr(dateStr);
      d.setDate(d.getDate() + days);
      return toDateStr(d);
    }

    function getMondayStr(dateStr) {
      const d = parseDateStr(dateStr);
      const day = d.getDay();
      const diff = day === 0 ? -6 : 1 - day;
      d.setDate(d.getDate() + diff);
      return toDateStr(d);
    }

    const todayStr = toDateStr(new Date());

    function changeWeek(deltaDays) {
      const nextMonday = addDays(state.currentMonday, deltaDays);

        alert('2026년~2027년 범위 내에서만 이동할 수 있습니다.');

    }

    function toggleDay(dayNum) {
      if (selectedDays.includes(dayNum)) {
        selectedDays = selectedDays.filter(d => d !== dayNum);
      } else {
        selectedDays.push(dayNum);
      }
      document.querySelectorAll('#dayPicker button').forEach(btn => {
        const d = parseInt(btn.dataset.day, 10);
        if (selectedDays.includes(d)) {
          btn.classList.add('active');
        } else {
          btn.classList.remove('active');
        }
      });
    }

    function getValidStudyDates(startStr, endStr, studyDaysArr) {
      const validDates = [];
      let curr = parseDateStr(startStr);
      const end = parseDateStr(endStr);
      while (curr <= end) {
        if (studyDaysArr.includes(curr.getDay())) {
          validDates.push(toDateStr(curr));
        }
        curr.setDate(curr.getDate() + 1);
      }
      return validDates;
    }

        return;
      }
      if (!startDate || !endDate || endDate < startDate) {
        alert('시작 날짜와 목표 완강일을 올바르게 선택해주세요.');
        return;
      }
      if (selectedDays.length === 0) {
        alert('진도를 나갈 요일을 최소 1개 이상 선택해주세요.');
        return;
      }

      const validDates = getValidStudyDates(startDate, endDate, selectedDays);
      if (validDates.length === 0) {
        alert('선택한 기간 내에 해당 공부 요일이 없습니다. 기간이나 요일을 조정해주세요.');
        return;
      }

      const totalCount = endLec - startLec + 1;
      const lectures = [];

      for (let i = 0; i < totalCount; i++) {
        const lecNum = startLec + i;
        const dateIdx = Math.floor((i * validDates.length) / totalCount);
        lectures.push({
          id: 'lec_' + Date.now() + '_' + i,
          num: lecNum,
          title: `${lecNum}강`,
          date: validDates[dateIdx],
          done: false
        });
      }

      state.courses.push({
        id: 'course_' + Date.now(),
        name,
        color: selectedColor,
        startLec,
        endLec,
        startDate,
        endDate,
        studyDays: [...selectedDays],
        lectures
      });

      state.currentMonday = getMondayStr(startDate);
      document.getElementById('courseName').value = '';
      document.getElementById('endLec').value = '';
      toggleModal(false);
      saveState();
    }

    function deleteCourse(courseId) {
      if (!confirm('이 과목과 모든 강의 계획을 삭제할까요?')) return;
      state.courses = state.courses.filter(c => c.id !== courseId);
      saveState();
    }

    function toggleLecture(courseId, lecId) {
      const course = state.courses.find(c => c.id === courseId);
      if (!course) return;
      const lec = course.lectures.find(l => l.id === lecId);
      if (lec) {
        lec.done = !lec.done;
        saveState();
      }
    }

    function delayOneDay(courseId, lecId, event) {
      event.stopPropagation();
      const course = state.courses.find(c => c.id === courseId);
      if (!course) return;
      const lec = course.lectures.find(l => l.id === lecId);
      if (lec) {
        lec.date = addDays(lec.date, 1);
        saveState();
      }
    }

    function moveMissedToSaturday() {
      const monStr = state.currentMonday;
      const friStr = addDays(monStr, 4);
      const satStr = addDays(monStr, 5);
      let count = 0;

      state.courses.forEach(course => {
        course.lectures.forEach(lec => {
          if (!lec.done && lec.date >= monStr && lec.date <= friStr) {
            lec.date = satStr;
            count++;
          }
        });
      });

      if (count === 0) {
        alert('이번 주 평일(월~금)에 밀린 미달성 강의가 없습니다!');
        return;
      }
      saveState();
      alert(`평일 미달성 강의 ${count}개를 이번 주 토요일(${satStr.slice(5)}) 보충 일정으로 모았습니다!`);
    }

    function pushWeekToNextMonday() {
      const monStr = state.currentMonday;
      const sunStr = addDays(monStr, 6);
      const nextMonStr = addDays(monStr, 7);
      let count = 0;

      state.courses.forEach(course => {
        course.lectures.forEach(lec => {
          if (!lec.done && lec.date >= monStr && lec.date <= sunStr) {
            lec.date = nextMonStr;
            count++;
          }
        });
      });

      if (count === 0) {
        alert('이번 주에 남아있는 미달성 강의가 없습니다!');
        return;
      }
      state.currentMonday = nextMonStr;
      saveState();
      alert(`이번 주 미완료 강의 ${count}개를 다음 주 월요일(${nextMonStr.slice(5)})로 이월했습니다.`);
    }

    function redistributeAll() {
      const baseDateStr = state.currentMonday;
      let count = 0;

      state.courses.forEach(course => {
        const remaining = course.lectures
          .filter(l => !l.done && l.date >= baseDateStr)
          .sort((a, b) => a.num - b.num);

        if (remaining.length === 0) return;

        const targetEnd = course.endDate > baseDateStr ? course.endDate : addDays(baseDateStr, 6);
        let validDates = getValidStudyDates(baseDateStr, targetEnd, course.studyDays || [1,2,3,4,5]);
        if (validDates.length === 0) validDates = [baseDateStr];

        remaining.forEach((lec, idx) => {
          const dIdx = Math.floor((idx * validDates.length) / remaining.length);
          lec.date = validDates[dIdx];
        });
        count++;
      });

      if (count === 0) {
        alert('재분배할 미완료 강의가 없습니다.');
        return;
      }
      saveState();
      alert('남은 미완료 강의를 목표 기간 내 공부 요일에 맞춰 일별로 균등 재분배했습니다!');
    }

    function render() {
      const courseListEl = document.getElementById('courseList');
      const dailyListEl = document.getElementById('dailyScheduleList');
      const weekRangeTitleEl = document.getElementById('weekRangeTitle');
      const weekProgressTextEl = document.getElementById('weekProgressText');
      const totalSummaryEl = document.getElementById('totalSummary');

      const monStr = state.currentMonday;
      const sunStr = addDays(monStr, 6);
      weekRangeTitleEl.textContent = `${monStr.replace(/-/g, '.')} (월) ~ ${sunStr.slice(5).replace(/-/g, '.')} (일)`;

      if (state.courses.length === 0) {
        courseListEl.innerHTML = `
          <div class="bg-white rounded-2xl p-5 text-center border border-dashed border-slate-300 text-slate-400 text-xs leading-relaxed">
            우측 상단 <b>[+ 과목 추가]</b>를 눌러<br>시작 강의 번호와 기간, 공부 요일을 설정해보세요!
          </div>`;
        totalSummaryEl.textContent = '';
      } else {
        let allTotal = 0;
        let allDone = 0;

        courseListEl.innerHTML = state.courses.map(course => {
          const total = course.lectures.length;
          const done = course.lectures.filter(l => l.done).length;
          const percent = total > 0 ? Math.round((done / total) * 100) : 0;
          const theme = colorMap[course.color] || colorMap.indigo;

          allTotal += total;
          allDone += done;

          return `
            <div class="bg-white rounded-2xl p-3.5 shadow-2xs border border-slate-200/80">
              <div class="flex justify-between items-center mb-1.5">
                <div class="flex items-center gap-1.5">
                  <span class="text-[11px] font-bold px-2 py-0.5 rounded-md ${theme.badge}">${course.startLec}~${course.endLec}강</span>
                  <h3 class="font-extrabold text-slate-900 text-sm">${course.name}</h3>
                </div>
                <div class="flex items-center gap-2">
                  <span class="text-xs font-extrabold text-slate-800">${percent}% <span class="text-[11px] font-normal text-slate-400">(${done}/${total})</span></span>
                  <button onclick="deleteCourse('${course.id}')" class="text-slate-300 hover:text-rose-500 text-xs">✕</button>
                </div>
              </div>
              <div class="w-full bg-slate-100 h-2 rounded-full overflow-hidden">
                <div class="${theme.bar} h-full transition-all duration-300 rounded-full" style="width: ${percent}%"></div>
              </div>
            </div>
          `;
        }).join('');

        const totalPercent = allTotal > 0 ? Math.round((allDone / allTotal) * 100) : 0;
        totalSummaryEl.textContent = `총 ${totalPercent}% (${allDone}/${allTotal}강)`;
      }

      let weekTotal = 0;
      let weekDone = 0;
      let daysHtml = '';
      const realToday = toDateStr(new Date());

      for (let i = 0; i < 7; i++) {
        const dStr = addDays(monStr, i);
        const dObj = parseDateStr(dStr);
        const dayIdx = dObj.getDay();
        const dayLabel = KOR_DAYS[dayIdx];
        const isSat = dayIdx === 6;
        const isSun = dayIdx === 0;
        const isToday = dStr === realToday;

        const dayItems = [];
        state.courses.forEach(course => {
          course.lectures.forEach(lec => {
            if (lec.date === dStr) {
              dayItems.push({ course, lec });
            }
          });
        });

        weekTotal += dayItems.length;
        const doneInDay = dayItems.filter(item => item.lec.done).length;
        weekDone += doneInDay;

        let badgeClass = 'bg-slate-100 text-slate-700';
        if (isSat) badgeClass = 'bg-amber-100 text-amber-800';
        if (isSun) badgeClass = 'bg-rose-100 text-rose-700';
        if (isToday) badgeClass = 'bg-indigo-600 text-white';

        daysHtml += `
          <div class="rounded-xl border ${isToday ? 'border-indigo-400 bg-indigo-50/20' : isSat ? 'border-amber-200/80 bg-amber-50/20' : 'border-slate-200/80 bg-slate-50/40'} p-3">
            <div class="flex justify-between items-center mb-2">
              <div class="flex items-center gap-1.5">
                <span class="text-xs font-extrabold px-2 py-0.5 rounded-lg ${badgeClass}">
                  ${dStr.slice(5).replace('-', '/')} (${dayLabel})
                </span>
                ${isToday ? '<span class="text-[11px] font-bold text-indigo-600">오늘</span>' : ''}
                ${isSat ? '<span class="text-[11px] font-semibold text-amber-700">보충 데이</span>' : ''}
              </div>
              <span class="text-[11px] font-semibold text-slate-400">
                ${dayItems.length > 0 ? `${doneInDay}/${dayItems.length} 완료` : '일정 없음'}
              </span>
            </div>

            ${dayItems.length === 0 ? `
              <p class="text-xs text-slate-300 py-1 pl-1">배정된 강의가 없습니다.</p>
            ` : `
              <div class="space-y-1.5">
                ${dayItems.map(({ course, lec }) => {
                  const theme = colorMap[course.color] || colorMap.indigo;
                  return `
                    <div onclick="toggleLecture('${course.id}', '${lec.id}')"
                      class="flex items-center justify-between p-2.5 rounded-xl border cursor-pointer transition ${
                        lec.done
                          ? 'bg-slate-100/70 border-slate-200 text-slate-400 line-through'
                          : 'bg-white border-slate-200/90 text-slate-800 shadow-2xs active:bg-slate-50'
                      }">
                      <div class="flex items-center gap-2.5 min-w-0">
                        <div class="w-5 h-5 rounded-md shrink-0 flex items-center justify-center border ${
                          lec.done ? `${theme.bar} border-transparent text-white` : 'border-slate-300 bg-white'
                        }">
                          ${lec.done ? '✓' : ''}
                        </div>
                        <span class="text-[11px] font-bold px-1.5 py-0.5 rounded ${theme.badge} shrink-0">${course.name}</span>
                        <span class="text-sm font-bold truncate">${lec.title}</span>
                      </div>
                      ${!lec.done ? `
                        <button onclick="delayOneDay('${course.id}', '${lec.id}', event)"
                          class="text-[11px] font-semibold text-slate-400 hover:text-indigo-600 bg-slate-100 px-2 py-1 rounded-lg shrink-0 ml-2">
                          +1일 미루기
                        </button>
                      ` : ''}
                    </div>
                  `;
                }).join('')}
              </div>
            `}
          </div>
        `;
      }

      dailyListEl.innerHTML = daysHtml;
      const wPercent = weekTotal > 0 ? Math.round((weekDone / weekTotal) * 100) : 0;
      weekProgressTextEl.textContent = `이번 주 ${weekDone} / ${weekTotal}강 완료 (${wPercent}%)`;
    }

    render();
  </script>
</body>
